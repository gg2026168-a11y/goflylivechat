input.addEventListener('input', () => {

验证方式：用 http 服务托管两个页面（别用 file://，否则 WebSocket 的 Origin 是 null 可能被拒），开两个标签：

· visitor.html?sid=1
· agent.html?sid=1

在访客端输入框打字，客服端上方粉色框会实时出现「访客正在输入：xxx」。访客端自己的消息区不会出现任何「正在输入」字样，因为它的 onmessage 里第一行就是 if (m.type !== 'chat_message') return;。

如果你实际跑起来看不到，最可能的原因是：

1. 两个页面 sid 不一致，不在同一个房间；
2. 页面用 file:// 打开，WebSocket 连不上；
3. 后端没启动或 DSN 写错，initDB 失败直接 log.Fatalf 退出了。

  let m;
  if (/Windows NT 10\.0/.test(ua)) return 'Windows 10/11';
  if (/Windows NT 6\.3/.test(ua))  return 'Windows 8.1';
  if (/Windows NT 6\.1/.test(ua))  return 'Windows 7';
  if (/Windows NT/.test(ua))       return 'Windows';
  if ((m = ua.match(/Android\s([\d.]+)/))) return 'Android ' + m[1];
  if ((m = ua.match(/iPhone OS ([\d_]+)/))) return 'iOS ' + m[1].replace(/_/g, '.');
  if ((m = ua.match(/iPad.*?OS ([\d_]+)/))) return 'iPadOS ' + m[1].replace(/_/g, '.');
  if ((m = ua.match(/Mac OS X ([\d_]+)/))) return 'macOS ' + m[1].replace(/_/g, '.');
  if (/CrOS/.test(ua))  return 'Chrome OS';
  if (/Ubuntu/.test(ua)) return 'Ubuntu';
  if (/Linux/.test(ua))  return 'Linux';
  return '未知系统';
}

function detectBrowser(ua) {
  let m;
  // 注意顺序：Edge / Opera 的 UA 中同样包含 Chrome 与 Safari
  if ((m = ua.match(/Edg(?:e|A|iOS)?\/([\d.]+)/))) return 'Edge ' + m[1].split('.')[0];
  if ((m = ua.match(/OPR\/([\d.]+)/)))              return 'Opera ' + m[1].split('.')[0];
  if ((m = ua.match(/Firefox\/([\d.]+)/)))          return 'Firefox ' + m[1].split('.')[0];
  if ((m = ua.match(/MicroMessenger\/([\d.]+)/)))   return '微信内置浏览器';
  if ((m = ua.match(/Chrome\/([\d.]+)/)))           return 'Chrome ' + m[1].split('.')[0];
  if ((m = ua.match(/Version\/([\d.]+).*Safari/)))  return 'Safari ' + m[1].split('.')[0];
  if (/Safari/.test(ua)) return 'Safari';
  return '未知浏览器';
}

function detectDeviceType(ua) {
  if (/iPad|Tablet|PlayBook|Silk/.test(ua)) return '平板电脑';
  if (/Android/.test(ua) && !/Mobile/.test(ua)) return '平板电脑';
  if (/Mobi|iPhone|iPod|Windows Phone|Android.*Mobile|HarmonyOS/.test(ua)) return '手机';
  return '桌面电脑';
}
