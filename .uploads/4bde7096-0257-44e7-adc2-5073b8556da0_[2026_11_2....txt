let querying = false;

async function verifyMobile() {
  if (querying) return;

  const raw = inputEl.value.trim();
  const mobile = normalizeMobile(raw);
  if (mobile && mobile !== raw) inputEl.value = mobile;

  if (!mobile) {
    inputEl.classList.remove('valid');
    inputEl.classList.add('invalid');
    setStatus('error', '请输入手机号码。');
    resetResult(); lockNext();
    return;
  }

  if (!isValidChinaMobile(mobile)) {
    inputEl.classList.remove('valid');
    inputEl.classList.add('invalid');
    setStatus('error', '号码格式不正确：中国大陆手机号应为 1 开头的 11 位数字。');
    resetResult(); lockNext();
    return;
  }

  inputEl.classList.remove('invalid', 'valid');
  resetResult(); lockNext();

  querying = true;
  setStatus('loading', '<span class="spinner"></span> 正在查询归属地，请稍候…');

  try {
    const data = await ip138Query(mobile);

    if (!data || data.ret !== 'ok') {
      const msg = data && data.msg ? data.msg : '接口未返回有效结果';
      throw new Error('NO_RESULT:' + msg);
    }

    const d = data.data;
    if (!Array.isArray(d)  d.length < 2  (!d[0] && !d[1])) {
      throw new Error('NO_RESULT:号码未收录，未查询到归属地信息');
    }

    const province = d[0] || '';
    const city     = d[1] || '';
    const carrier  = d[2] || '';
    const zone     = d[3] || '';

    if (!province && !city) {

    }

    inputEl.classList.add('valid');
    $('rProvince').textContent = province || '—';
    $('rCity').textContent     = city || '—';
    $('rCarrier').textContent  = carrier || '—';
    $('rZone').textContent     = zone || '—';
    resultEl.classList.add('show');

    setStatus('success', 核验通过：${province || '—'} ${city || ''} · ${carrier || '运营商未知'});
    unlockNext();

  } catch (err) {
    inputEl.classList.remove('valid');
    inputEl.classList.add('invalid');
    resetResult(); lockNext();

    const msg = err && err.message ? err.message : '';

    if (msg.startsWith('NO_RESULT:')) {
      setStatus('error', '查询无果：' + msg.replace('NO_RESULT:', '') + '。无法进行下一步。');
    } else if (msg === 'TIMEOUT') {
      setStatus('error', '查询超时：ip138 接口未在规定时间内响应，请检查网络后重试。');

  }
}
